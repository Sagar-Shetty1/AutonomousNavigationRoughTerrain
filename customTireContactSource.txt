function [Fx,Fy,Fz] = RoverTireModel_FL(vx,vy,rho,rhoDot,omega)

kz = 5000;
cz = 300;
mu = 0.7;

R  = 0.076;   % Wheel radius (m)
Cx = 300;     % Initial longitudinal slip coefficient

Fz = max(0, kz*rho + cz*rhoDot);

% Longitudinal slip velocity due to wheel rotation
vSlip = R*omega - vx;

% Traction force generated by wheel slip
Fx = Cx*vSlip;

% Retain the simplified lateral resistance
Fy = -1500*vy;

% Limit tangential forces by available friction
Fx = max(-mu*Fz, min(mu*Fz, Fx));
Fy = max(-mu*Fz, min(mu*Fz, Fy));

end
